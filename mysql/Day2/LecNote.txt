======= Sql : 
department : create table department(dep_Number integer (20) PRIMARY KEY AUTO_INCREMENT, name varchar (255) unique NOT Null);

department_location : create table department_location(dep_number_fk integer (20)  not null , location varchar (255) ,
foreign key (dep_number_fk) REFERENCES department (dep_Number)  ON DELETE CASCADE ON UPDATE CASCADE);


employee :CREATE TABLE employee (
    ssn INT PRIMARY KEY AUTO_INCREMENT,
    address VARCHAR(150),
    name VARCHAR(200),
    BD DATE,
    gender ENUM('male', 'female'),
    salary DECIMAL(7,3),

    dep_num_fk INT NULL,
    emp_supervisor INT  NULL,

    FOREIGN KEY (dep_num_fk)
        REFERENCES department(dep_Number)
        ON DELETE SET NULL
        ON UPDATE CASCADE,

    FOREIGN KEY (emp_supervisor)
        REFERENCES employee(ssn)
        ON DELETE SET NULL
        ON UPDATE CASCADE
);


project :CREATE TABLE project (
    number integer(10) primary key auto_increment,
    name varchar(255) unique  not null , 
    location varchar(255) , 
    dep_number_fk integer (20) ,
    foreign key (dep_number_fk) REFERENCES department (dep_Number)  ON DELETE  set null ON UPDATE CASCADE
);

Emp_project :CREATE TABLE Emp_project (
    id INT PRIMARY KEY AUTO_INCREMENT,
    emp_num INT NULL,
    pro_num INT  NULL,
    hours decimal(3,2),

      foreign key (pro_num) 
      REFERENCES project  (number) 
       ON DELETE  set null 
       ON UPDATE CASCADE,

        FOREIGN KEY (emp_num)
        REFERENCES employee(ssn)
        ON DELETE SET NULL
        ON UPDATE CASCADE

);



Emp_dependences :CREATE TABLE Emp_dependences (
   
    id INT (20) primary Key,
   name VARCHAR(200),
    BD DATE,
    gender ENUM('male', 'female'),
    emp_num int not null ,

        FOREIGN KEY (emp_num)
        REFERENCES employee(ssn)
        ON DELETE cascade
        ON UPDATE CASCADE

);

==================== select =======
all data :select * from department

--------------- condition ------

select * from employee where salary>1000

-----------------range---------
select * from employee where salary between 1000 and 2000;


select * from employee where address in ("cairo","banha","sadat");

              --------------------- NOT IN -------------
select * from employee where address Not in ("cairo","banha","sadat");

---------- null data ----
SELECT * FROM `employee` WHERE address is null;
---------- not null data ----
SELECT * FROM `employee` WHERE address is NOT null;


--------------------- Select DISTINCT : slect data that not repeated --

select DISTINCT address from employee;

----------- And , OR , Not
SELECT * FROM `employee` WHERE not salary = 7000;
SELECT * FROM `employee` WHERE  salary = 7000 and address="cairo";
SELECT * FROM `employee` WHERE  salary = 7000 or address="cairo";

================= select using order by 

max to low : SELECT * FROM `employee` ORDER by salary DESC;


low to max : SELECT * FROM `employee` ORDER by salary ASC ;==>default


========== Aggregiation Functions : max , min , sum , count , avg
SELECT SUM(salary) AS "total salary" FROM employee;
SELECT MIN(salary) AS "min salary" FROM employee;
SELECT Max(salary) AS "max salary" FROM employee;
SELECT AVG(salary) AS "Avg salary" FROM employee;
