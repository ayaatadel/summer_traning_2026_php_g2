FROM department d LEFT join Employee e
 ON d.dep_Number = e.dep_num_fk
  GROUP BY d.dep_Number, d.name;

  -------------------------------------------- select ALL DATA IN department untill (employee = null)
SELECT d.dep_Number, d.name, COUNT(e.ssn) AS employee_count 
FROM department d Right join Employee e
 ON d.dep_Number = e.dep_num_fk
  GROUP BY d.dep_Number, d.name;

  -------------------------------------------- select ALL DATA IN employee  untill (department = null)
SELECT d.dep_Number, d.name, COUNT(e.ssn) AS employee_count 
FROM department d inner join Employee e
 ON d.dep_Number = e.dep_num_fk
  GROUP BY d.dep_Number, d.name;

  -------------------------------------------- select data shared between employee and department
SELECT d.dep_Number, d.name, COUNT(e.ssn) AS employee_count 
FROM department d  join Employee e
 ON d.dep_Number = e.dep_num_fk
  GROUP BY d.dep_Number, d.name;

  full join (left union right )

SELECT d.dep_Number, d.name, COUNT(e.ssn) AS employee_count FROM department d left join Employee e ON d.dep_Number = e.dep_num_fk GROUP BY d.dep_Number, d.name UNION SELECT d.dep_Number, d.name, COUNT(e.ssn) AS employee_count FROM department d RIGHT join Employee e ON d.dep_Number = e.dep_num_fk GROUP BY d.dep_Number, d.name;
